// "CRT_FILEX"
struct CRT_FILEX
{
	struct CRT_iobuf file;

};

// "CRT_MBCPInfo"

{
	uint32_t codePage;
	uint32_t mbulinfo[0x3];
	uint8_t ranges[0x4][0x8];
};

// "CRT_RuntimeErrorMessage"
struct CRT_RuntimeErrorMessage
{
	int32_t error_number;
	char const* message;
};

// "CRT_iobuf"
struct CRT_iobuf
{
	char* _ptr;
	int32_t _cnt;
	char* _base;
	int32_t _flag;
	int32_t _file;
	int32_t _charbuf;
	int32_t _bufsiz;
	char* _tmpfname;
};

// "CRT_ioinfo"

{

	char pipech;
	char pad_06[0x2];
